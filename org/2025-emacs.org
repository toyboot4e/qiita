*** =org-qiita-export-buffer-to-book= の作成

後は =.org= ファイルからの Markdown ファイルの展開を実装します。幸い org ドキュメントの /subtree/ に対する /export/ という概念があり、 org-mode はこのような処理が得意です。

#+ATTR_TYPST: :breakable true
#+BEGIN_SRC elisp
(leaf ox-qmd
  :commands (org-qiita-export-buffer-to-book)
  :config
  ;; Converts org-mode tags into front matter tag.
  (defun qmd2-template-tags ()
    (let ((tags (or (mapcar #'substring-no-properties (org-get-tags))
                    "")))
      (string-join
       (mapcar (lambda (s) (format "\"%s\"" s)) tags)
       ",")))

  ;; Markdown file template for `@qiita/qiita-cli'.
  (defun qmd2-template (contents info)
    (let* ((title (or (substring-no-properties (car (plist-get info :title))) ""))
           (tags (qmd2-template-tags)))
      (concat
       "---\n"
       "title: \"" title "\"\n"
       "tags: [" tags "]\n"
       "private: false\n"
       "updated_at: \"\"\n"
       "id: null\n"
       "organization_url_name: null\n"
       "slide: false\n"
       "---\n\n"
       contents)))

  ;; The Qiita Markdown export backend, built on top of `ox-qmd'.
  (org-export-define-derived-backend 'qmd2 'qmd
    :options-alist
    '((:title "TITLE" nil nil t)
      (:tags "TAGS" nil nil space))
    :translate-alist '((template . qmd2-template)))

  ;; Exports a subtree of org file into a Markdown file.
  (defun org-qmd2-export-to-markdown (&optional async subtreep visible-only)
    (interactive)
    (let ((outfile (org-export-output-file-name ".md" subtreep)))
      (org-export-to-file 'qmd2 outfile async subtreep visible-only)))

  ;; Exports each level 1 heading into a Markdown file.
  (defun org-qiita-export-buffer-to-book (&optional org-dir)
    "Runs subtree export to each level 1 heading, respecting `#+BOOK_DIR'."
    (interactive)
    (require 'ox-qmd)
    (let* ((default-dir default-directory)
           (pub-dir (car (cdr (car (org-collect-keywords '("BOOK_DIR")))))))
      (cd pub-dir)
      (unwind-protect
          (org-map-entries
           (lambda ()
             (let ((exports (org-entry-get nil "EXPORT_FILE_NAME")))
               (when exports
                 (org-qmd2-export-to-markdown nil t))))
           "LEVEL=1")
        (cd default-dir)))))
#+END_SRC

#+BEGIN_SRC emacs-lisp
    (progn ;; Show full path in `find-file' after `embark-become'
      (defun my-embark-become-prepend-default-dir (orig-fun command input)
        "Prepend default-directory to INPUT when COMMAND is file-related."
        (if (and (memq command '(find-file find-file-other-window
                                           find-file-other-frame find-file-literally))
                 (not (string-prefix-p (expand-file-name default-directory)
                                       (expand-file-name input))))
            (let ((dir-with-slash (file-name-as-directory default-directory)))
              (if (equal input "")
                  (funcall orig-fun command dir-with-slash)
                (funcall orig-fun command (expand-file-name input dir-with-slash))))
          (funcall orig-fun command input)))

      (advice-add 'embark--become-command :around
                  #'my-embark-become-prepend-default-dir)))
#+END_SRC

*** =this-command= について

=embark= は minibuffer 突入時の =this-command= の値を =embark--command= に記録しています。 =embark-become= ではこの値を見て、何のコマンドから移動しようとしているか判定します。

** Bookmark を使用する

*** Mark とは

Mark はカーソル位置 (_point_) を表すデータです。 [[https://www.gnu.org/software/emacs/manual/html_node/emacs/Mark-Ring.html][Mark ring]] が普段からカーソル位置の変化を記録しており、 =consult-mark= で確認できます。 Mark は一時的なデータであり、 Emacs の終了時に失われます。

*** Bookmark とは

[[https://www.gnu.org/software/emacs/manual/html_node/emacs/Bookmarks.html][Bookmarks]] はファイル保存される mark のことだと思います。 Builtin の bookmark はフラットな一覧であるため、これらにタグを付けてフィルタリングして行きたいです。

*** Bookmark, Bookmark+

[[https://www.emacswiki.org/emacs/BookmarkPlus][Bookmark+]] は Emacs Wiki で開発・公開されているパッケージです。最新版にこだわりがなければ、 [[https://github.com/emacsmirror/bookmark-plus][GitHub のミラー]] から引っ張ってくるのが簡単そうです:

#+BEGIN_SRC elisp
(leaf bookmark+
  :ensure nil
  :vc (:url "https://github.com/emacsmirror/bookmark-plus"))
#+END_SRC

#+BEGIN_EXPORT md
:::note warn
設定不足で恐縮です。

Bookmark ファイルの更新は、 Emacs の終了時などまで遅延されます。 Bookmark の一時的な状態がメモリ上のみに存在するため、 Emacs が異常終了した場合にデータが失われる他、複数の Emacs 間の同期が取れません。

また Bookmark の変更の度にファイル保存する設定もありますが、 `bookmark+` の関数 (`bmkp-*`) のパフォーマンスが極端に落ちたため未設定です。
:::
#+END_EXPORT

*** 実装

**** Bookmark 保存

Bookmark を以下の形で保存したいです:

- 名称: =dir/file.exe:line= (自動設定)
- タグ: =repository/topic= (トピック部分はユーザ指定)

これらの計算用の関数です:

#+BEGIN_SRC elisp
(defun my/get-default-bookmark-name ()
  "dir/file.exe:line"
  (let* ((root (vc-git-root default-directory))
         (line (line-number-at-pos)))
    (format "%s:%d" (file-relative-name (buffer-file-name) root) line)))

(defun my/project-like-name ()
  "git root name or a fallback"
  (file-name-nondirectory
   (directory-file-name
    (or (vc-git-root default-directory)
        (directory-file-name default-directory)
        (buffer-name)))))
#+END_SRC

これを元にカーソル位置を bookmark として保存するコマンドを作成しました:

#+BEGIN_SRC elisp
(defun my/bookmark-set ()
  "Create a bookmark at point with automatic file-based naming and auto tag."
  (interactive)
  (let ((bookmark-name (my/get-default-bookmark-name))
        (tag-name (consult-bookmark-tags-with-default "Tag: " t)))
    (let ((inhibit-message t))
      (bookmark-set bookmark-name nil t)
      (bmkp-add-tags bookmark-name `(,tag-name)))
    (message "Bookmark created: %s (tag: %s)" bookmark-name tag-name)))
#+END_SRC

**** Bookmark を確認

Bookmark をタグ付きで確認するため、 =consult-bookmark= の candidates を書き換えてみました:

#+BEGIN_SRC elisp
(defun my/consult--bookmark-candidates ()
  "Return bookmark candidates."
  (require 'bookmark+)
  (cl-loop for bm in bookmark-alist collect
           (let* ((name (car bm))
                  (tags (bmkp-get-tags bm))
                  (line-content (let ((filename (bookmark-get-filename bm))
                                      (pos (bookmark-get-position bm)))
                                  (cond
                                   ((and filename pos (file-exists-p filename) (not (file-directory-p filename)))
                                    (with-temp-buffer
                                      (insert-file-contents filename)
                                      (goto-char pos)
                                      (string-trim (thing-at-point 'line t))))
                                   ((and filename (file-directory-p filename))
                                    (format "[Directory: %s]" (file-name-nondirectory filename)))
                                   (filename
                                    (format "[File: %s]" (file-name-nondirectory filename)))
                                   (t "[Buffer]"))))
                  (tags-str (if tags
                                (propertize (mapconcat #'identity tags ", ") 'face 'consult-key)
                              ""))
                  ;; We're embedding nerd icon here..
                  (display-name (format "%-30s %-20s %s" (concat (nerd-icons-icon-for-file name) " " name) tags-str
                                        (propertize (or line-content "") 'face 'completions-annotations))))
             (propertize display-name
                         'consult--candidate name))))
#+END_SRC

最悪ですが、 =consult-bookmark= をコピペして、上の candidates を使ってみます:

#+BEGIN_SRC elisp
(defun my/consult-bookmark-with-tags (name)
  "`consult-bookmark' with another candidates definition"
  (interactive
   (list
    (require 'consult)
    (bookmark-maybe-load-default-file)
    (setq bookmark-alist (my/bookmark-sort bookmark-alist))
    (let ((narrow (cl-loop for (x y . _) in consult-bookmark-narrow collect (cons x y))))
      (consult--read
       (my/consult--bookmark-candidates)
       :prompt "Bookmark: "
       :state (consult--bookmark-preview)
       :category 'bookmark
       :history 'bookmark-history
       :initial (or my/bookmark-auto-tag "")
       ;; Add default names to future history.
       ;; Ignore errors such that `consult-bookmark' can be used in
       ;; buffers which are not backed by a file.
       :add-history (ignore-errors (bookmark-prop-get (bookmark-make-record) 'defaults))
       :group (consult--type-group narrow)
       :narrow (consult--type-narrow narrow)
       :lookup #'consult--lookup-candidate))))
  (bookmark-maybe-load-default-file)
  (if (assoc name bookmark-alist)
      (bookmark-jump name)
    (bookmark-set name)))
