#+BEGIN_SRC sqlite
create table hello_a (msg TEXT, a INTEGER);
#+END_SRC

#+RESULTS:

#+BEGIN_SRC sqlite
.table
#+END_SRC

#+RESULTS:
: hello    hello_a

#+BEGIN_SRC sqlite
insert into hello_a (msg, a) VALUES ('hello', 42);
insert into hello_a (msg, a) VALUES ('hello_world', 43);
#+END_SRC

#+RESULTS:

#+BEGIN_SRC sqlite
select * from hello;
select * from hello_a;
#+END_SRC

#+RESULTS:
| hello       |    |
| hello       | 42 |
| hello_world | 43 |

Inner join:

#+BEGIN_SRC sqlite
SELECT * FROM hello
INNER JOIN hello_a
ON hello_a.msg == hello.msg;
#+END_SRC

#+RESULTS:
| hello | hello | 42 |

Right outer join:

#+BEGIN_SRC sqlite
SELECT * FROM hello
RIGHT JOIN hello_a
ON hello_a.msg = hello.msg;
#+END_SRC

#+RESULTS:
| hello | hello       | 42 |
|       | hello_world | 43 |
