#+BEGIN_SRC sqlite

#+END_SRC

#+RESULTS:

#+BEGIN_SRC sqlite
select * from hello;
select * from hello_a;
#+END_SRC

#+RESULTS:
| hello |    |
| hello | 42 |

Inner join:

#+BEGIN_SRC sqlite
SELECT * FROM hello
INNER JOIN hello_a
ON hello_a.msg == hello.msg;
#+END_SRC

#+RESULTS:
| hello | hello | 42 |

Left outer join:

#+BEGIN_SRC sqlite
SELECT * FROM hello
LEFT JOIN hello_a
ON hello_a.msg = hello.msg;
#+END_SRC

#+RESULTS:
| hello | hello | 42 |
